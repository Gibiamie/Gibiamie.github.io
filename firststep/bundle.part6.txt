s: renderFavorites,
    privacy: renderPrivacy
  };
  main.innerHTML = (routes[state.route] || renderHome)();

  window.scrollTo({ top: 0, behavior: 'instant' });
  bindPageEvents();
}

function renderSpontaneous() {
  const { person, anchor, detail, variant } = state.spontaneous;
  const needsDetail = anchor === 'queue';
  const ready = Boolean(anchor) && (!needsDetail || Boolean(detail));
  const result = ready ? spontaneousSuggestion({ person, anchor, detail, lang: state.lang, variant }) : null;

  const anchors = [
    ['ordering', t('ordering'), '☕'],
    ['queue', t('queue'), '↔'],
    ['live_music', t('live_music'), '♫'],
    ['event', t('event'), '✦'],
    ['screen', t('screen'), '▣'],
    ['book', t('book'), '▤'],
    ['venue_advice', t('venue_advice'), '⌂'],
    ['no_anchor', t('no_anchor'), '—']
  ];

  return `${header(t('spontaneous'), t('spontaneousDesc'))}
    <section class="panel compact">
      <h2>${t('person')}</h2>
      <div class="chip-row">
        ${chip('single', t('single'), person === 'single', 'sp-person')}
        ${chip('pair', t('pair'), person === 'pair', 'sp-person')}
        ${chip('group', t('group'), person === 'group', 'sp-person')}
      </div>
    </section>
    <section class="panel compact">
      <h2>${t('anchor')}</h2>
      <div class="choice-grid quick anchor-grid">
        ${anchors.map(([value, label, symbol]) => quickChoice(value, label, symbol, anchor === value, 'anchor')).join('')}
      </div>
    </section>
    ${needsDetail ? `<section class="panel compact">
      <h2>${t('queueDetail')}</h2>
      <div class="choice-grid queue-detail-grid">
        ${quickChoice('order_service', t('order_service'), '☕', detail === 'order_service', 'queue-detail')}
        ${quickChoice('entrance', t('entrance'), '⇥', detail === 'entrance', 'queue-detail')}
        ${quickChoice('transport', t('transport_wait'), '⌁', detail === 'transport', 'queue-detail')}
        ${quickChoice('event_entry', t('event_entry'), '✦', detail === 'event_entry', 'queue-detail')}
      </div>
    </section>` : ''}
    ${result ? `<section class="result-card instant-result ${result.kind === 'guidance' ? 'guidance-result' : ''}">
      <div class="result-label ${result.kind === 'guidance' ? 'warning' : ''}">${result.kind === 'guidance' ? t('noOpener') : t('sayNow')}</div>
      <blockquote>${escapeHtml(result.text)}</blockquote>
      ${result.kind === 'opener' ? `<div class="result-actions">
        <button class="secondary-button" data-action="sp-next">${icon('refresh')} ${t('warmer')}</button>
        <button class="secondary-button" data-action="save" data-save-text="${escapeHtml(result.text)}" data-save-meta="${escapeHtml(t('spontaneous'))}">${icon('heart')} ${t('save')}</button>
      </div>` : ''}
      <p class="safety-line">${t('groundingNotice')}</p>
    </section>` : ''}`;
}

function quickChoice(value, label, symbol, selected = false, kind = 'anchor') {
  const attr = kind === 'queue-detail' ? `data-queue-detail="${value}"` : `data-anchor="${value}"`;
  return `<button class="quick-choice ${selected ? 'selected' : ''}" ${attr}><span>${symbol}</span><strong>${escapeHtml(label)}</strong></button>`;
}
