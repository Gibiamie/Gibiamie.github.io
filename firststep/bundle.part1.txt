function labelForVenueType(type, lang = 'tr') {
  return labels[lang]?.[type] || labels.tr[type] || type;
}

function choose(items, seed = 0) {
  if (!items.length) return '';
  return items[Math.abs(seed) % items.length];
}

function simpleSeed(...parts) {
  return parts.join('|').split('').reduce((sum, ch) => sum + ch.charCodeAt(0), 0);
}

function timeBand(date = new Date()) {
  const hour = date.getHours();
  if (hour < 11) return 'morning';
  if (hour < 17) return 'afternoon';
  if (hour < 22) return 'evening';
  return 'night';
}
