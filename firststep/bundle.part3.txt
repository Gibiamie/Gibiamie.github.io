function contextualSuggestion({
  type = 'other', venueName = '', location = '', group = 'single', activity = 'eating', openness = 'neutral', lang = 'tr', variant = 0
}) {
  if (openness === 'closed') {
    return lang === 'tr'
      ? 'Şu anda yaklaşmamak en saygılı seçenek. Doğal bir fırsat oluşursa kısa bir selam yeterlidir.'
      : 'Not approaching now is the most respectful option. If a natural opportunity appears, a brief greeting is enough.';
  }
  if (openness === 'busy') {
    return lang === 'tr'
      ? 'Doğal bir ara oluşmasını bekleyin. Ara oluşmazsa konuşmayı başlatmayın.'
      : 'Wait for a natural pause. If no pause appears, do not start the conversation.';
  }

  const base = venueOpeners[lang]?.[type] || venueOpeners.tr.other;
  const seed = simpleSeed(type, venueName, location, group, activity, variant);
  let text = choose(base, seed + variant);

  if (activity === 'reading' && group === 'single') {
    text = lang === 'tr'
      ? 'Merhaba, elinizdeki kitabı gördüm. Okuduğunuz kadarıyla nasıl buldunuz?'
      : 'Hi, I noticed the book you are holding. How are you finding it so far?';
  } else if (activity === 'ordering') {
    text = lang === 'tr'
      ? 'Merhaba, menüde kararsız kaldım. Denediğiniz bir şey varsa tavsiye eder misiniz?'
      : 'Hi, I am deciding between the menu options. If you have tried something here, would you recommend it?';
  } else if (activity === 'eating') {
    text = lang === 'tr'
      ? 'Merhaba, sipariş ettiğiniz şeyi nasıl buldunuz? Tavsiye eder misiniz?'
      : 'Hi, how is what you ordered? Would you recommend it?';
  } else if (activity === 'live_music') {
    text = lang === 'tr'
      ? 'Merhaba, sahnedeki grubu daha önce dinlediniz mi?'
      : 'Hi, have you heard the band on stage before?';
  } else if (activity === 'screen') {
    text = lang === 'tr'
      ? 'Merhaba, ekrandaki yayını siz de takip ediyor musunuz?'
      : 'Hi, are you following the broadcast on the screen too?';
  } else if (activity === 'shared_event') {
    text = lang === 'tr'
      ? 'Merhaba, etkinliği şu ana kadar nasıl buldunuz?'
      : 'Hi, how are you finding the event so far?';
  } else if (activity === 'phone') {
    text = lang === 'tr'
      ? 'Kişi telefonla ilgileniyor. Doğal bir ara oluşmadan konuşmayı başlatmayın.'
      : 'The person is using their phone. Do not start until a natural pause appears.';
  } else if (activity === 'group_talking') {
    text = lang === 'tr'
      ? 'Grup aktif biçimde konuşuyor. Ortak bir olay veya doğal bir ara olmadıkça konuşmaya girmeyin.'
      : 'The group is actively talking. Do not enter unless there is a shared event or a natural pause.';
  }

  return text;
}
