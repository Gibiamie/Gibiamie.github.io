function escapeHtml(value = '') {
  return String(value).replace(/[&<>'"]/g, (ch) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[ch]));
}

function chip(value, label, selected, group, extra = '') {
  return `<button type="button" class="chip ${selected ? 'selected' : ''}" data-chip-group="${group}" data-value="${value}" ${extra}>${escapeHtml(label)}</button>`;
}

function saveFavorite(text, meta = '') {
  if (!text) return;
  if (!state.favorites.some((item) => item.text === text)) {
    state.favorites.unshift({ id: Date.now(), text, meta, createdAt: new Date().toISOString() });
    state.favorites = state.favorites.slice(0, 30);
    localStorage.setItem('fs_favorites', JSON.stringify(state.favorites));
  }
  toast(t('saved'));
}

function toast(message) {
  const el = $('#toast');
  el.textContent = message;
  el.classList.add('show');
  clearTimeout(toast.timer);
  toast.timer = setTimeout(() => el.classList.remove('show'), 2200);
}
