const insufficientContextText = {
  tr: 'Güvenilir bir ortak bağlam seçilmedi. Zorlanmış veya anlamsız bir cümle üretmek yerine kısa bir selam için doğal bir an bekleyin.',
  en: 'No reliable shared context was selected. Instead of generating a forced or vague line, wait for a natural moment for a brief greeting.'
};

function spontaneousSuggestion({ person = 'single', anchor = null, detail = null, lang = 'tr', variant = 0 }) {
  if (!anchor || anchor === 'no_anchor') {
    return { text: insufficientContextText[lang] || insufficientContextText.tr, kind: 'guidance' };
  }

  if (anchor === 'queue' && !detail) {
    return {
      text: lang === 'tr'
        ? 'Sıranın ne için olduğunu seçmeden güvenilir bir cümle üretilemez.'
        : 'A reliable opener cannot be generated until you identify what the queue is for.',
      kind: 'guidance'
    };
  }

  const block = spontaneousTemplates[lang]?.[anchor] || spontaneousTemplates.tr[anchor];
  const options = anchor === 'queue' ? block?.[detail] : block;
  if (!Array.isArray(options) || !options.length) {
    return { text: insufficientContextText[lang] || insufficientContextText.tr, kind: 'guidance' };
  }

  // Person/group selection is retained for later tone adaptation. It must not fabricate facts.
  const offset = person === 'group' ? 1 : person === 'pair' ? 2 : 0;
  return { text: options[(variant + offset) % options.length], kind: 'opener' };
}
