function renderRescue() {
  const { situation, variant } = state.rescue;
  const suggestion = situation ? rescueSuggestion({ situation, lang: state.lang, variant }) : '';
  const alt = situation ? rescueSuggestion({ situation, lang: state.lang, variant: variant + 1 }) : '';
  const situations = ['silence', 'short_answer', 'asks_question', 'topic_ended', 'interested', 'leaving', 'joined_group'];
  return `${header(t('rescueTitle'), t('rescueSubtitle'))}
    <section class="choice-grid rescue-grid">
      ${situations.map((item) => `<button class="quick-choice ${situation === item ? 'selected' : ''}" data-rescue="${item}"><span>${rescueSymbol(item)}</span><strong>${t(item)}</strong></button>`).join('')}
    </section>
    ${suggestion ? `<section class="result-card instant-result">
      <div class="result-label">${t('doThis')}</div>
      <blockquote>${escapeHtml(suggestion)}</blockquote>
      <div class="alternative"><span>${t('alternative')}</span><p>${escapeHtml(alt)}</p></div>
      <div class="result-actions">
        <button class="secondary-button" data-action="rescue-next">${icon('refresh')} ${t('warmer')}</button>
        <button class="secondary-button" data-action="save" data-save-text="${escapeHtml(suggestion)}" data-save-meta="${escapeHtml(t('rescue'))}">${icon('heart')} ${t('save')}</button>
      </div>
    </section>` : ''}`;
}

function rescueSymbol(item) {
  return ({ silence: '…', short_answer: '—', asks_question: '?', topic_ended: '↪', interested: '＋', leaving: '↗', joined_group: '◎' })[item] || '•';
}

function renderPractice() {
  const scenarios = practiceScenarios[state.lang];
  const selectedScenario = state.practice.scenario ? scenarios.find((s) => s.id === state.practice.scenario) : null;
  if (!selectedScenario) {
    return `${header(t('practiceTitle'), t('chooseScenario'))}
      <section class="scenario-list">${scenarios.map((scenario) => `<button class="scenario-card" data-scenario="${scenario.id}"><span>${icon('practice')}</span><div><strong>${escapeHtml(scenario.title)}</strong><small>${escapeHtml(scenario.scene)}</small></div>${icon('arrow')}</button>`).join('')}</section>`;
  }

  const chosen = state.practice.selected;
  return `${header(selectedScenario.title, selectedScenario.scene)}
    <section class="panel">
      <h2>${t('chooseOpener')}</h2>
      <div class="answer-list">${selectedScenario.choices.map((choice, index) => `<button class="answer ${chosen === index ? 'selected' : ''} ${chosen !== null && index === selectedScenario.best ? 'best' : ''}" data-practice-choice="${index}">${escapeHtml(choice)}</button>`).join('')}</div>
    </section>
    ${chosen !== null ? `<section class="result-card practice-result">
      <div class="result-label ${chosen === selectedScenario.best ? 'success' : 'warning'}">${chosen === selectedScenario.best ? t('correct') : t('improve')}</div>
      <div class="mini-section"><span>${t('response')}</span><p>“${escapeHtml(selectedScenario.response)}”</p></div>
      <div class="mini-section"><span>${t('feedback')}</span><p>${escapeHtml(selectedScenario.feedback)}</p></div>
      <button class="primary-button full" data-action="practice-reset">${t('retry')}</button>
    </section>` : ''}`;
}
